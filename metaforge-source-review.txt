// METAFORGE-ADOBE-KEYWORDS-SEARCH
import { randomUUID } from "node:crypto";
import { isKnownGeminiModel } from "../../../lib/providers/gemini-models";
import { generateStrictMetadata } from "../../../lib/server/generateStrictMetadata";
import { getCurrentUser, saveHistory } from "../../../lib/server/history";
import { checkRateLimit } from "../../../lib/server/rateLimit";
import { cleanFileName, formField, isSameOrigin, jsonError, readUploadedImage } from "../../../lib/server/requestUtils";

export async function POST(request: Request): Promise<Response> {
  const requestId = randomUUID();
  if (!isSameOrigin(request)) return jsonError(403, "forbidden", "This request was not allowed.", false, requestId);
  const client = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "local";
  const limit = checkRateLimit(`adobe-keywords:${client}`, 30, 60_000);
  if (!limit.allowed) return jsonError(429, "rate_limited", "Too many requests. Please wait a moment and try again.", true, requestId, { "Retry-After": String(limit.retryAfterSeconds) });
  const who = await getCurrentUser();
  if (who.configured && !who.userId) return jsonError(401, "unauthenticated", "Please sign in.", false, requestId);
  let form: FormData;
  try { form = await request.formData(); } catch { return jsonError(400, "invalid_request", "The upload could not be read.", false, requestId); }
  const model = formField(form, "model");
  if (!isKnownGeminiModel(model)) return jsonError(400, "unsupported_model", "Choose a supported Gemini model.", false, requestId);
  const upload = await readUploadedImage(form, requestId);
  if (!upload.ok) return upload.response;
  const generated = await generateStrictMetadata({ model, mimeType: upload.mimeType, base64: Buffer.from(upload.bytes).toString("base64"), count: 50 });
  if (!generated.ok) return jsonError(generated.status, generated.code, generated.message, generated.retryable, requestId, generated.retryAfterSeconds !== null ? { "Retry-After": String(generated.retryAfterSeconds) } : undefined);
  const seen = new Set<string>();
  const keywords: string[] = [];
  for (const raw of generated.data.keywords) { const word = raw.trim(); const key = word.toLocaleLowerCase(); if (word && !seen.has(key)) { seen.add(key); keywords.push(word); } }
  if (keywords.length !== 50) return jsonError(502, "keyword_validation_failed", `The model returned ${keywords.length} unique keywords instead of 50. Please try again.`, true, requestId);
  const data = { title: generated.data.title, description: generated.data.description, keywords, warnings: generated.warnings };
  let saved = false;
  if (who.userId) saved = await saveHistory(who.userId, { kind: "metadata", fileName: cleanFileName(formField(form, "fileName")), provider: "gemini", model, platform: "adobe-stock", result: data });
  return Response.json({ success: true, data, provider: "gemini", model, saved, requestId }, { headers: { "Cache-Control": "no-store" } });
}

import { adminApiGuard } from "../../../lib/auth/session";
import { MAX_KEYS_PER_PROVIDER, PROVIDERS, type ProviderId } from "../../../lib/providers/catalog";
import { validateApiKey, validateLabel } from "../../../lib/keys/shared";
import { addKey, getStorageState, listKeys, removeKey, setEnabled } from "../../../lib/server/keyStore";

// Responses NEVER include a saved secret. Only masked values are returned.
// Every request needs a signed-in admin (checked on the server).

const ID_PATTERN = /^[0-9a-fA-F-]{36}$/;

function reply(body: Record<string, unknown>, status = 200): Response {
  return Response.json(body, { status, headers: { "Cache-Control": "no-store" } });
}

function fail(status: number, code: string, message: string): Response {
  return reply({ success: false, code, message }, status);
}

function isSameOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");
  if (!origin) {
    return true;
  }
  try {
    const host = request.headers.get("x-forwarded-host") ?? request.headers.get("host");
    return host !== null && new URL(origin).host === host;
  } catch {
    return false;
  }
}

async function guard(request: Request): Promise<Response | null> {
  if (!isSameOrigin(request)) {
    return fail(403, "forbidden", "This request was not allowed.");
  }
  const client = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "local";
  const limit = checkRateLimit(`api-keys:${client}`, 120, 60_000);
  if (!limit.allowed) {
    return fail(429, "rate_limited", "Too many requests. Please wait a moment.");
  }
  return adminApiGuard();
}

async function readBody(request: Request): Promise<Record<string, unknown> | null> {
  try {
    const value: unknown = await request.json();
    return typeof value === "object" && value !== null && !Array.isArray(value)
      ? (value as Record<string, unknown>)
      : null;
  } catch {
    return null;
  }
}

function asProviderId(value: unknown): ProviderId | null {
  return typeof value === "string" && PROVIDERS.some((provider) => provider.id === value)
    ? (value as ProviderId)
    : null;
}

export async function GET(request: Request): Promise<Response> {
  const blocked = await guard(request);
  if (blocked) {
    return blocked;
  }

  const storage = getStorageState();
  if (storage.state !== "ready") {
    return reply({ success: true, storage, limit: MAX_KEYS_PER_PROVIDER, keys: [] });
  }

  try {
    const keys = await listKeys();
    return reply({ success: true, storage, limit: MAX_KEYS_PER_PROVIDER, keys });
  } catch {
    return reply({
      success: true,
      storage: {
        state: "error",
        message:
          "The saved keys could not be read. Check that the api_keys SQL was run in Supabase and that SUPABASE_SERVICE_ROLE_KEY is correct.",
      },
      limit: MAX_KEYS_PER_PROVIDER,
      keys: [],
    });
  }
}

export async function POST(request: Request): Promise<Response> {
  const blocked = await guard(request);
  if (blocked) {
    return blocked;
  }

  const body = await readBody(request);
  if (!body) {
    return fail(400, "invalid_request", "The request could not be read.");
  }

  const provider = asProviderId(body.provider);
  if (!provider) {
    return fail(400, "invalid_request", "Unknown provider.");
  }
  if (typeof body.label !== "string" || typeof body.apiKey !== "string") {
    return fail(400, "invalid_request", "A label and an API key are required.");
  }

  const problem = validateLabel(body.label) ?? validateApiKey(body.apiKey);
  if (problem) {
    return fail(400, "validation_failed", problem);
  }

  try {
    const result = await addKey({
      provider,
      label: body.label.trim(),
      apiKey: body.apiKey.trim(),
    });
    if (!result.ok) {
      return fail(result.status, result.code, result.message);
    }
    return reply({ success: true, key: result.value }, 201);
  } catch {
    return fail(500, "storage_error", "The key could not be saved. Check the Supabase setup and try again.");
  }
}

export async function PATCH(request: Request): Promise<Response> {
  const blocked = await guard(request);
  if (blocked) {
    return blocked;
  }

  const body = await readBody(request);
  if (!body || typeof body.id !== "string" || !ID_PATTERN.test(body.id) || typeof body.enabled !== "boolean") {
    return fail(400, "invalid_request", "A key id and an enabled value are required.");
  }

  try {
    const result = await setEnabled(body.id, body.enabled);
    if (!result.ok) {
      return fail(result.status, result.code, result.message);
    }
    return reply({ success: true, key: result.value });
  } catch {
    return fail(500, "storage_error", "The key could not be updated. Try again.");
  }
}

export async function DELETE(request: Request): Promise<Response> {
  const blocked = await guard(request);
  if (blocked) {
    return blocked;
  }

  const id = new URL(request.url).searchParams.get("id") ?? "";
  if (!ID_PATTERN.test(id)) {
    return fail(400, "invalid_request", "A valid key id is required.");
  }

  try {
    const result = await removeKey(id);
    if (!result.ok) {
      return fail(result.status, result.code, result.message);
    }
    return reply({ success: true });
  } catch {
    return fail(500, "storage_error", "The key could not be removed. Try again.");
  }
}

export async function GET(request: Request): Promise<Response> {
  const who = await getCurrentUser();
  if (!who.configured) {
    return fail(503, "auth_not_configured", "History needs sign-in and the database, which are not configured.");
  }
  if (!who.userId) {
    return fail(401, "unauthenticated", "Please sign in.");

    if (error instanceof HistoryError) {
      return fail(500, "history_unavailable", "History could not be read. Check that the history SQL was run in Supabase.");

    return fail(500, "internal_error", "Something went wrong. Please try again.");

function greetingFor(hour: number) {
  if (hour >= 5 && hour < 12) return "Good Morning";

function initials(name: string) {
  const parts = name.trim().split(/\s+/).filter(Boolean);

}

function safeHttpsUrl(value: string): string | null {
  if (!value) return null;
  try {
    const u = new URL(value);
    return u.protocol === "https:" ? u.toString() : null;
  } catch {
    return null;
  }
}
const SUPPORT_URL = safeHttpsUrl(SUPPORT_URL_RAW);

async function copyText(text: string): Promise<boolean> {
  try {
    if (navigator.clipboard && window.isSecureContext) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch {
    // fall through to the legacy method
  }
  try {
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.setAttribute("readonly", "");
    ta.style.position = "fixed";
    ta.style.opacity = "0";
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand("copy");
    document.body.removeChild(ta);
    return ok;
  } catch {
    return false;
  }
}

// Contains no API keys, tokens, query strings, file contents or history.
function buildDiagnostics(): string {
  let prefs = "unknown";
  try {
    prefs =
      window.localStorage.getItem("metaforge.preferences.v1") !== null
        ? "saved in this browser"
        : "not saved (defaults in use)";
  } catch {
    prefs = "local storage unavailable";
  }
  return [
    "METAFORGE AI PRO - troubleshooting information",
    "Generated locally in your browser. Nothing was sent anywhere.",
    "Time: " + new Date().toISOString(),
    "Page: " + window.location.pathname,
    "Browser: " + navigator.userAgent,
    "Viewport: " + window.innerWidth + "x" + window.innerHeight,
    "Online: " + (navigator.onLine ? "yes" : "no"),
    "Saved settings: " + prefs,
    "Not included: API keys, tokens, passwords, file contents, history.",
  ].join("\n");
}

import { randomUUID } from "node:crypto";
import { getAccess } from "../../../lib/auth/session";
import { isKnownGeminiModel } from "../../../lib/providers/gemini-models";
import { generateStrictMetadata } from "../../../lib/server/generateStrictMetadata";
import { saveHistory } from "../../../lib/server/history";
import { checkRateLimit } from "../../../lib/server/rateLimit";
import { cleanFileName, formField, isSameOrigin, jsonError, readUploadedImage } from "../../../lib/server/requestUtils";

export async function POST(request: Request): Promise<Response> {
  const requestId = randomUUID();
  if (!isSameOrigin(request)) return jsonError(403, "forbidden", "This request was not allowed.", false, requestId);
  const client = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "local";
  const limit = checkRateLimit(`adobe-keywords:${client}`, 30, 60_000);
  if (!limit.allowed) return jsonError(429, "rate_limited", "Too many requests. Please wait a moment and try again.", true, requestId, { "Retry-After": String(limit.retryAfterSeconds) });

  // The user id comes only from the server-verified session, never from the request.
  const access = await getAccess();
  if (!access.ok) return jsonError(access.status, access.code, access.message, access.status === 503, requestId);

  let form: FormData;
  try { form = await request.formData(); } catch { return jsonError(400, "invalid_request", "The upload could not be read.", false, requestId); }
  const model = formField(form, "model");
  if (!isKnownGeminiModel(model)) return jsonError(400, "unsupported_model", "Choose a supported Gemini model.", false, requestId);
  const upload = await readUploadedImage(form, requestId);
  if (!upload.ok) return upload.response;
  const generated = await generateStrictMetadata({ userId: access.userId, model, mimeType: upload.mimeType, base64: Buffer.from(upload.bytes).toString("base64"), count: 50 });
  if (!generated.ok) return jsonError(generated.status, generated.code, generated.message, generated.retryable, requestId, generated.retryAfterSeconds !== null ? { "Retry-After": String(generated.retryAfterSeconds) } : undefined);
  const seen = new Set<string>();
  const keywords: string[] = [];
  for (const raw of generated.data.keywords) { const word = raw.trim(); const key = word.toLocaleLowerCase(); if (word && !seen.has(key)) { seen.add(key); keywords.push(word); } }
  if (keywords.length !== 50) return jsonError(502, "keyword_validation_failed", `The model returned ${keywords.length} unique keywords instead of 50. Please try again.`, true, requestId);
  const data = { title: generated.data.title, description: generated.data.description, keywords, warnings: generated.warnings };
  const saved = await saveHistory(access.userId, { kind: "metadata", fileName: cleanFileName(formField(form, "fileName")), provider: "gemini", model, platform: "adobe-stock", result: data });
  return Response.json({ success: true, data, provider: "gemini", model, saved, requestId }, { headers: { "Cache-Control": "no-store" } });
}

type GuardResult =
  | { ok: true; userId: string }
  | { ok: false; response: Response };

function reply(
  body: Record<string, unknown>,
  status = 200,
): Response {
  return Response.json(body, {
    status,
    headers: {
      "Cache-Control": "no-store, private",
      Pragma: "no-cache",
      "X-Content-Type-Options": "nosniff",

function fail(
  status: number,
  code: string,
  message: string,
): Response {
  return reply({ success: false, code, message }, status);

function isSameOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");

  // Write requests must include a valid Origin header.
  if (!origin) return false;

  try {
    const originUrl = new URL(origin);
    const forwardedHost = request.headers
      .get("x-forwarded-host")
      ?.split(",")[0]
      ?.trim();

    const requestHost =
      forwardedHost ||
      request.headers.get("host");

    if (!requestHost) return false;

    return (
      originUrl.host.toLowerCase() === requestHost.toLowerCase() &&
      (originUrl.protocol === "https:" ||
        originUrl.protocol === "http:")
    );
  } catch {
    return false;
  }
}

async function guard(
  request: Request,
  requireSameOrigin = false,
): Promise<GuardResult> {
  if (requireSameOrigin && !isSameOrigin(request)) {
    return {
      ok: false,
      response: fail(
        403,
        "forbidden",
        "This request was not allowed.",
      ),
    };
  }

  const client =
    request.headers
      .get("x-forwarded-for")
      ?.split(",")[0]
      ?.trim() || "local";

  const limit = checkRateLimit(
    `api-keys:${client}`,
    120,
    60_000,
  );

  if (!limit.allowed) {
    return {
      ok: false,
      response: fail(
        429,
        "rate_limited",
        "Too many requests. Please wait a moment.",
      ),
    };
  }

  try {
    const access = await getAccess();

    if (!access.ok) {
      return {
        ok: false,
        response: fail(
          access.status,
          access.code,
          access.message,
        ),
      };
    }

    return { ok: true, userId: access.userId };
  } catch {
    return {
      ok: false,
      response: fail(
        500,
        "auth_error",
        "Authentication could not be verified. Please try again.",
      ),
    };
  }
}

async function readBody(
  request: Request,
): Promise<Record<string, unknown> | null> {
  try {
    const value: unknown = await request.json();

    if (
      typeof value !== "object" ||
      value === null ||
      Array.isArray(value)
    ) {
      return null;
    }

    return value as Record<string, unknown>;
  } catch {
    return null;
  }
}

export async function POST(
  request: Request,
): Promise<Response> {

  if (!checked.ok) return checked.response;

  const body = await readBody(request);

  if (!body) {
    return fail(
      400,
      "invalid_request",
      "The request body is invalid.",
    );
  }

  const provider = asProviderId(body.provider);

  if (!provider) {
    return fail(
      400,
      "invalid_request",
      "Unknown API provider.",
    );
  }

  if (
    typeof body.label !== "string" ||
    typeof body.apiKey !== "string"
  ) {
    return fail(
      400,

      "A label and API key are required.",
    );
  }

  const label = body.label.trim();
  const apiKey = body.apiKey.trim();

  const problem =
    validateLabel(label) ??
    validateApiKey(apiKey);

  if (problem) {
    return fail(400, "validation_failed", problem);

      provider,
      label,
      apiKey,
    });

      return fail(
        result.status,

import { execFile, type ExecFileException } from "node:child_process";
import { mkdtemp, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import path from "node:path";
import { checkRateLimit } from "../../../lib/server/rateLimit";

// Renders an EPS to a JPEG with Ghostscript. Needs Ghostscript installed on the server.
// Hosted platforms such as Vercel do not include it. Access still goes through the
// sign-in proxy, and the request must come from this site.

const MAX_BYTES = 8 * 1024 * 1024;
const TIMEOUT_MS = 30_000;
const MAX_OUTPUT_BYTES = 8 * 1024 * 1024;

type GsOutcome = "ok" | "missing" | "timeout" | "failed";

function reply(status: number, code: string, message: string): Response {
  return Response.json(
    { success: false, code, message },
    { status, headers: { "Cache-Control": "no-store" } },
  );
}

function isSameOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");
  if (!origin) {
    return true;
  }
  try {
    const host = request.headers.get("x-forwarded-host") ?? request.headers.get("host");
    return host !== null && new URL(origin).host === host;
  } catch {
    return false;
  }
}

function looksLikeEps(bytes: Uint8Array): boolean {
  if (bytes.length >= 4 && bytes[0] === 0xc5 && bytes[1] === 0xd0 && bytes[2] === 0xd3 && bytes[3] === 0xc6) {
    return true;
  }
  return Buffer.from(bytes.subarray(0, 11)).toString("latin1") === "%!PS-Adobe-";
}

// Keeps the longest side near 3000 pixels so huge artboards cannot exhaust memory.
function chooseDpi(buffer: Buffer): number {
  const head = buffer.subarray(0, 131072).toString("latin1");
  const match = /%%BoundingBox:\s*(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)/.exec(head);
  if (!match) {
    return 100;
  }
  const width = Math.abs(Number(match[3]) - Number(match[1]));
  const height = Math.abs(Number(match[4]) - Number(match[2]));
  const longest = Math.max(width, height, 1);
  return Math.max(20, Math.min(150, Math.floor((3000 * 72) / longest)));
}

function candidates(): string[] {
  const configured = (process.env.GHOSTSCRIPT_PATH ?? "").trim();
  const defaults = process.platform === "win32" ? ["gswin64c", "gswin32c", "gs"] : ["gs"];
  return configured ? [configured, ...defaults] : defaults;
}

function runGhostscript(command: string, args: string[]): Promise<GsOutcome> {
  return new Promise((resolve) => {
    execFile(
      command,
      args,
      { timeout: TIMEOUT_MS, killSignal: "SIGKILL", windowsHide: true, maxBuffer: 1024 * 1024 },
      (error: ExecFileException | null) => {
        if (!error) {
          resolve("ok");
        } else if (error.code === "ENOENT") {
          resolve("missing");
        } else if (error.killed) {
          resolve("timeout");
        } else {
          resolve("failed");
        }
      },
    );
  });
}

export async function POST(request: Request): Promise<Response> {
  if (!isSameOrigin(request)) {
    return reply(403, "forbidden", "This request was not allowed.");
  }

  const client = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "local";
  if (!checkRateLimit(`eps:${client}`, 30, 60_000).allowed) {
    return reply(429, "rate_limited", "Too many conversions. Please wait a moment.");
  }

  let form: FormData;
  try {
    form = await request.formData();
  } catch {
    return reply(400, "invalid_request", "The upload could not be read.");
  }

  const file = form.get("file");
  if (file === null || typeof file === "string") {
    return reply(400, "invalid_request", "No file was uploaded.");
  }
  if (file.size === 0) {
    return reply(400, "invalid_request", "The file is empty.");
  }
  if (file.size > MAX_BYTES) {
    return reply(413, "payload_too_large", "The EPS is larger than 8 MB.");
  }

  const bytes = Buffer.from(await file.arrayBuffer());
  if (!looksLikeEps(bytes)) {
    return reply(400, "invalid_file", "This is not a valid EPS file.");
  }

  const dir = await mkdtemp(path.join(tmpdir(), "metaforge-eps-"));
  try {
    const input = path.join(dir, "input.eps");
    const output = path.join(dir, "output.jpg");
    await writeFile(input, bytes);

    const args = [
      "-dSAFER",
      "-dBATCH",
      "-dNOPAUSE",
      "-dQUIET",
      "-dEPSCrop",
      "-sDEVICE=jpeg",
      "-dJPEGQ=88",
      "-dGraphicsAlphaBits=4",
      "-dTextAlphaBits=4",
      `-r${chooseDpi(bytes)}`,
      `-sOutputFile=${output}`,
      input,
    ];

    let outcome: GsOutcome = "missing";
    for (const command of candidates()) {
      outcome = await runGhostscript(command, args);
      if (outcome !== "missing") {
        break;
      }
    }

    if (outcome === "missing") {
      return reply(
        503,
        "renderer_missing",
        "EPS conversion needs Ghostscript, which is not installed on this server. Install Ghostscript and restart, or set GHOSTSCRIPT_PATH. Hosted platforms such as Vercel do not include it.",
      );
    }
    if (outcome === "timeout") {
      return reply(504, "timeout", "Converting this EPS took too long.");
    }
    if (outcome === "failed") {
      return reply(422, "conversion_failed", "This EPS could not be rendered. It may be damaged or use unsupported features.");
    }

    let jpeg: Buffer;
    try {
      jpeg = await readFile(output);
    } catch {
      return reply(422, "conversion_failed", "This EPS produced no image.");
    }
    if (jpeg.length === 0 || jpeg.length > MAX_OUTPUT_BYTES) {
      return reply(422, "conversion_failed", "The converted image was empty or too large.");
    }

    return new Response(new Uint8Array(jpeg), {
      headers: { "Content-Type": "image/jpeg", "Cache-Control": "no-store" },
    });
  } catch {
    return reply(500, "internal_error", "The conversion failed unexpectedly.");
  } finally {
    await rm(dir, { recursive: true, force: true });
  }
}

import { HISTORY_PAGE_SIZE, HistoryError, deleteHistory, getCurrentUser, listHistory, type HistoryKind } from "../../../lib/server/history";

const ID_PATTERN = /^[0-9a-fA-F-]{36}$/;

function reply(body: Record<string, unknown>, status = 200): Response {
  return Response.json(body, { status, headers: { "Cache-Control": "no-store" } });
}

function fail(status: number, code: string, message: string): Response {
  return reply({ success: false, code, message }, status);
}

function isSameOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");
  if (!origin) {
    return true;
  }
  try {
    const host = request.headers.get("x-forwarded-host") ?? request.headers.get("host");
    return host === null ? true : new URL(origin).host === host;
  } catch {
    return false;
  }
}

export async function GET(request: Request): Promise<Response> {
  const who = await getCurrentUser();
  if (!who.configured) {
    return fail(503, "auth_not_configured", "History needs sign-in and the database, which are not configured.");
  }
  if (!who.userId) {
    return fail(401, "unauthenticated", "Please sign in.");
  }

  const params = new URL(request.url).searchParams;
  const page = Number(params.get("page") ?? "1");
  if (!Number.isInteger(page) || page < 1 || page > 1000) {
    return fail(400, "invalid_request", "Invalid page number.");
  }
  const kindParam = params.get("kind");
  if (kindParam !== null && kindParam !== "metadata" && kindParam !== "prompt") {
    return fail(400, "invalid_request", "Invalid history type.");
  }
  const kind: HistoryKind | null = kindParam;

  try {
    const { entries, total } = await listHistory(who.userId, { kind, page });
    return reply({ success: true, entries, total, page, pageSize: HISTORY_PAGE_SIZE });
  } catch (error) {
    if (error instanceof HistoryError) {
      return fail(500, "history_unavailable", "History could not be read. Check that the history SQL was run in Supabase.");
    }
    return fail(500, "internal_error", "Something went wrong. Please try again.");
  }
}

export async function DELETE(request: Request): Promise<Response> {
  if (!isSameOrigin(request)) {
    return fail(403, "forbidden", "This request was not allowed.");
  }

  const who = await getCurrentUser();
  if (!who.configured) {
    return fail(503, "auth_not_configured", "History needs sign-in and the database, which are not configured.");
  }
  if (!who.userId) {
    return fail(401, "unauthenticated", "Please sign in.");
  }

  const id = new URL(request.url).searchParams.get("id") ?? "";
  if (!ID_PATTERN.test(id)) {
    return fail(400, "invalid_request", "A valid entry id is required.");
  }

  try {
    const removed = await deleteHistory(who.userId, id);
    if (!removed) {
      return fail(404, "not_found", "That entry no longer exists.");
    }
    return reply({ success: true });
  } catch {
    return fail(500, "history_unavailable", "The entry could not be deleted. Try again.");
  }
}

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";
import { getSupabaseEnv } from "../../../lib/supabase/env";

// Signs the person out. The redirect response is created first and the cookie
// removals are written onto it directly, so the browser always receives them.

export async function POST(request: NextRequest): Promise<Response> {
  const origin = new URL(request.url).origin;
  const response = NextResponse.redirect(`${origin}/login`, 303);
  response.headers.set("Cache-Control", "no-store");

  const env = getSupabaseEnv();
  if (env) {
    const supabase = createServerClient(env.url, env.publishableKey, {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            response.cookies.set(name, value, options);
          });
        },
      },
    });

    try {
      // Revokes the session on Supabase and queues the cookie removal.
      await supabase.auth.signOut();
    } catch {
      // The safety net below still clears the browser's cookies.
    }
  }

  // Safety net: expire every Supabase session cookie, including chunked ones.
  for (const cookie of request.cookies.getAll()) {
    if (cookie.name.startsWith("sb-")) {
      response.cookies.set(cookie.name, "", { path: "/", maxAge: 0 });
    }
  }

  return response;
}

===== src\components\metadata\vector.ts =====


 // Safe SVG handling: validate SVG markup, rasterize to JPEG, and never load external DTDs.
const MAX_SVG_BYTES = 5 * 1024 * 1024;
const FORBIDDEN_TAGS = new Set([
  "script",
  "foreignobject",
  "iframe",
  "object",
  "embed",
  "audio",
  "video",
]);

export function isSvgFile(file: File): boolean {
  return /\.svg$/i.test(file.name);
}

function fail(file: File, reason: string): Error {
  return new Error(`${file.name}: ${reason}`);
}

function parseSize(value: string | null): number | null {
  if (!value || value.trim().endsWith("%")) return null;

  const number = Number.parseFloat(value);
  return Number.isFinite(number) && number > 0 ? number : null;
}

/**
 * Accept only the standard external SVG 1.1 DOCTYPE used by many vector files.
 * Remove it before parsing so the browser never needs to resolve an external DTD.
 * Unknown DOCTYPE declarations and all ENTITY declarations remain forbidden.
 */
function sanitizeSvgDeclaration(file: File, source: string): string {
  let clean = source.replace(/^\uFEFF/, "");

  if (/<!ENTITY\b/i.test(clean)) {
    throw fail(file, "unsupported SVG (entity declarations are not allowed).");
  }

  if (/<!DOCTYPE\b/i.test(clean)) {
    const standardSvgDoctype =
      /<!DOCTYPE\s+svg\s+PUBLIC\s+(["'])-\/\/W3C\/\/DTD SVG 1\.1\/\/EN\1\s+(["'])https?:\/\/www\.w3\.org\/Graphics\/SVG\/1\.1\/DTD\/svg11\.dtd\2\s*>/i;

    const match = clean.match(standardSvgDoctype);

    if (!match) {
      throw fail(
        file,
        "unsupported SVG (only the standard SVG 1.1 DOCTYPE is accepted).",
      );
    }

    clean = clean.replace(match[0], "");

    if (/<!DOCTYPE\b|<!ENTITY\b/i.test(clean)) {
      throw fail(file, "unsupported SVG (additional XML declarations are not allowed).");
    }
  }

  return clean;
}

function hasUnsafeCssUrl(value: string): boolean {
  const urls = value.matchAll(/url\(\s*(['"]?)(.*?)\1\s*\)/gi);

  for (const match of urls) {
    const target = (match[2] ?? "").trim();

    if (
      !target.startsWith("#") &&
      !/^data:image\/(?:png|jpeg|gif|webp);base64,/i.test(target)
    ) {
      return true;
    }
  }

  return false;
}

function checkElementSafety(file: File, root: Element): void {
  const all = [root, ...Array.from(root.querySelectorAll("*"))];

  for (const element of all) {
    const tag = element.localName.toLowerCase();

    if (FORBIDDEN_TAGS.has(tag)) {
      throw fail(file, `unsafe SVG content (<${tag}> is not allowed).`);
    }

    if (tag === "style") {
      const css = element.textContent ?? "";

      if (/@import\b/i.test(css) || hasUnsafeCssUrl(css)) {
        throw fail(file, "unsafe SVG content (external resource in a style).");
      }
    }

    for (const attribute of Array.from(element.attributes)) {
      const name = attribute.name.toLowerCase();
      const value = attribute.value.trim();

      if (name.startsWith("on")) {
        throw fail(file, "unsafe SVG content (event handler attribute).");
      }

      if (/javascript\s*:/i.test(value)) {
        throw fail(file, "unsafe SVG content (script link).");
      }

      if (name === "href" || name === "xlink:href") {
        const isLocalReference = value.startsWith("#");
        const isRasterData =
          /^data:image\/(?:png|jpeg|gif|webp);base64,/i.test(value);

        if (!isLocalReference && !isRasterData) {
          throw fail(file, "unsafe SVG content (external link or resource).");
        }
      }

      if (name === "style" && hasUnsafeCssUrl(value)) {
        throw fail(file, "unsafe SVG content (external resource in a style).");
      }
    }
  }
}

/** Validate SVG markup and set explicit pixel dimensions. */
function prepareSvgText(
  file: File,
  text: string,
  maxEdge: number,
): string {
  if (!Number.isFinite(maxEdge) || maxEdge < 1) {
    throw fail(file, "invalid image conversion size.");
  }

  const clean = sanitizeSvgDeclaration(file, text);
  const doc = new DOMParser().parseFromString(clean, "image/svg+xml");

  if (
    doc.documentElement.localName.toLowerCase() === "parsererror" ||
    doc.querySelector("parsererror")
  ) {
    throw fail(file, "this is not a valid SVG (the XML is malformed).");
  }

  const root = doc.documentElement;

  if (
    root.localName.toLowerCase() !== "svg" ||
    root.namespaceURI !== "http://www.w3.org/2000/svg"
  ) {
    throw fail(file, "this is not a valid SVG image.");
  }

  checkElementSafety(file, root);

  let width = parseSize(root.getAttribute("width"));
  let height = parseSize(root.getAttribute("height"));

  const viewBoxValues = (root.getAttribute("viewBox") ?? "")
    .trim()
    .split(/[\s,]+/)
    .map(Number);

  const hasViewBox =
    viewBoxValues.length === 4 &&
    viewBoxValues.every(Number.isFinite) &&
    viewBoxValues[2] > 0 &&
    viewBoxValues[3] > 0;

  if (hasViewBox) {
    const viewWidth = viewBoxValues[2];
    const viewHeight = viewBoxValues[3];

    if (width === null || height === null) {
      width = viewWidth;
      height = viewHeight;
    }
  } else if (width === null || height === null) {
    throw fail(file, "the SVG has no usable width, height or viewBox.");
  } else {
    root.setAttribute("viewBox", `0 0 ${width} ${height}`);
  }

  const originalWidth = width as number;
  const originalHeight = height as number;
  const scale = Math.min(
    maxEdge / Math.max(originalWidth, originalHeight),
    1,
  );

  root.setAttribute(
    "width",
    String(Math.max(1, Math.round(originalWidth * scale))),
  );
  root.setAttribute(
    "height",
    String(Math.max(1, Math.round(originalHeight * scale))),
  );

  return new XMLSerializer().serializeToString(root);
}

export async function rasterizeSvg(
  file: File,
  maxEdge: number,
  quality: number,
): Promise<Blob> {
  if (file.size === 0) {
    throw fail(file, "the file is empty.");
  }

  if (file.size > MAX_SVG_BYTES) {
    throw fail(file, "the SVG is larger than 5 MB.");
  }

  const safeQuality = Math.max(0, Math.min(1, quality));
  const svgText = prepareSvgText(file, await file.text(), maxEdge);
  const blobUrl = URL.createObjectURL(
    new Blob([svgText], { type: "image/svg+xml;charset=utf-8" }),
  );

  try {
    const image = new Image();

    await new Promise<void>((resolve, reject) => {
      image.onload = () => resolve();
      image.onerror = () =>
        reject(fail(file, "the browser could not draw this SVG."));
      image.src = blobUrl;
    });

    const width = image.naturalWidth;
    const height = image.naturalHeight;

    if (
      width < 1 ||
      height < 1 ||
      width > maxEdge ||
      height > maxEdge
    ) {
      throw fail(file, "the converted SVG dimensions are invalid.");
    }

    const canvas = document.createElement("canvas");
    canvas.width = width;
    canvas.height = height;

    const context = canvas.getContext("2d");

    if (!context) {
      throw fail(file, "this browser could not prepare the image.");
    }

    // JPEG has no transparency, so use a white background.
    context.fillStyle = "#ffffff";
    context.fillRect(0, 0, width, height);
    context.drawImage(image, 0, 0, width, height);

    const output = await new Promise<Blob | null>((resolve) => {
      canvas.toBlob(resolve, "image/jpeg", safeQuality);
    });

    if (!output) {
      throw fail(file, "the SVG could not be converted to JPEG.");
    }

    return output;
  } finally {
    URL.revokeObjectURL(blobUrl);
  }
}
