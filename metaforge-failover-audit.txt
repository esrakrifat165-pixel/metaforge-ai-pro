import { createCipheriv, createDecipheriv, randomBytes, randomUUID, scryptSync } from "node:crypto";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { MAX_KEYS_PER_PROVIDER, PROVIDERS, type ProviderId } from "../providers/catalog";
import type { LastResult, PublicKey, StorageState } from "../keys/shared";
import { getSupabaseEnv } from "../supabase/env";

// SERVER ONLY. Never import this file from a client component.
//
// Keys are encrypted with AES-256-GCM (secret: KEY_ENCRYPTION_SECRET) and stored in the
// Supabase table public.api_keys. The table has row-level security and no policies, so
// only the server, using SUPABASE_SERVICE_ROLE_KEY, can read or write it.
//
// Every operation takes the verified owner ID of the signed-in user and filters by
// owner_id. The caller must get that ID from the server-side session, never from the browser.

type EncryptedSecret = { iv: string; tag: string; data: string };

type PublicRow = {
  id: string;
  provider: ProviderId;
  label: string;
  last4: string;
  enabled: boolean;
  created_at: string;
  last_result: LastResult;
  last_used_at: string | null;
};

type SecretRow = PublicRow & { secret: EncryptedSecret };

export type KeyResult<T> =
  | { ok: true; value: T }
  | { ok: false; code: string; message: string; status: number };

export type Candidate = { id: string; label: string; apiKey: string };

export type CandidateResult =
  | { ok: true; source: "store" | "env"; candidates: Candidate[] }
  | {
      ok: false;
      code: "no_enabled_key" | "not_configured" | "key_store_error" | "unauthenticated";
      message: string;
    };

export class KeyStoreError extends Error {}

const TABLE = "api_keys";
const PUBLIC_COLUMNS = "id,provider,label,last4,enabled,created_at,last_result,last_used_at";
const ENV_NAMES: Partial<Record<ProviderId, string[]>> = {
  gemini: ["GEMINI_API_KEY", "GOOGLE_API_KEY"],
};
const RECENT_FAILURE_MS = 2 * 60 * 1000;
const OWNER_ID_PATTERN =
  /^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$/;

/* ---------- Guards ---------- */

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function isEncrypted(value: unknown): value is EncryptedSecret {
  return (
    isRecord(value) &&
    typeof value.iv === "string" &&
    typeof value.tag === "string" &&
    typeof value.data === "string"
  );
}

function isLastResult(value: unknown): value is LastResult {
  return value === null || value === "ok" || value === "rejected" || value === "rate_limited";
}

function isPublicRow(value: unknown): value is PublicRow {
  return (
    isRecord(value) &&
    typeof value.id === "string" &&
    typeof value.provider === "string" &&
    PROVIDERS.some((provider) => provider.id === value.provider) &&
    typeof value.label === "string" &&
    typeof value.last4 === "string" &&
    typeof value.enabled === "boolean" &&
    typeof value.created_at === "string" &&
    isLastResult(value.last_result) &&
    (value.last_used_at === null || typeof value.last_used_at === "string")
  );
}

function isSecretRow(value: unknown): value is SecretRow {
  return isRecord(value) && isEncrypted(value.secret) && isPublicRow(value);
}

function isOwnerId(value: unknown): value is string {
  return typeof value === "string" && OWNER_ID_PATTERN.test(value);
}

function notSignedIn(): { ok: false; code: string; message: string; status: number } {
  return { ok: false, code: "unauthenticated", message: "Please sign in.", status: 401 };
}

/* ---------- Configuration ---------- */

function getSecret(): string | null {
  const value = (process.env.KEY_ENCRYPTION_SECRET ?? "").trim();
  return value.length >= 32 ? value : null;
}

function getServiceKey(): string | null {
  const value = (
    process.env.SUPABASE_SERVICE_ROLE_KEY ??
    process.env.SUPABASE_SECRET_KEY ??
    ""
  ).trim();
  return value.length > 0 ? value : null;
}

export function getStorageState(): StorageState {
  const missing: string[] = [];
  if (!getSupabaseEnv()) {
    missing.push("NEXT_PUBLIC_SUPABASE_URL and NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY");
  }
  if (!getServiceKey()) {
    missing.push("SUPABASE_SERVICE_ROLE_KEY");
  }
  if (!getSecret()) {
    missing.push("KEY_ENCRYPTION_SECRET (at least 32 characters)");
  }
  if (missing.length > 0) {
    return {
      state: "missing_secret",
      message: `Key storage is not configured. Missing: ${missing.join(", ")}. Add them to .env.local (local) or to Vercel Environment Variables (production), then restart or redeploy.`,
    };
  }
  return {
    state: "ready",
    message:
      "Keys are encrypted on the server and stored in your Supabase database. They are never sent back to the browser.",
  };
}

let cachedClient: SupabaseClient | null = null;

function db(): SupabaseClient {
  const env = getSupabaseEnv();
  const serviceKey = getServiceKey();
  if (!env || !serviceKey) {
    throw new KeyStoreError("Key storage is not configured.");
  }
  if (!cachedClient) {
    cachedClient = createClient(env.url, serviceKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });
  }
  return cachedClient;
}

/* ---------- Encryption ---------- */

let keyCache: { secret: string; key: Buffer } | null = null;

function getCryptoKey(): Buffer | null {
  const secret = getSecret();
  if (!secret) {
    return null;
  }
  if (keyCache && keyCache.secret === secret) {
    return keyCache.key;
  }
  const key = scryptSync(secret, "metaforge-api-keys-v1", 32);
  keyCache = { secret, key };
  return key;
}

function encrypt(plain: string, aad: string, key: Buffer): EncryptedSecret {
  const iv = randomBytes(12);
  const cipher = createCipheriv("aes-256-gcm", key, iv);
  cipher.setAAD(Buffer.from(aad, "utf8"));
  const data = Buffer.concat([cipher.update(plain, "utf8"), cipher.final()]);
  return {
    iv: iv.toString("base64"),
    tag: cipher.getAuthTag().toString("base64"),
    data: data.toString("base64"),
  };
}

function decrypt(secret: EncryptedSecret, aad: string, key: Buffer): string | null {
  try {
    const decipher = createDecipheriv("aes-256-gcm", key, Buffer.from(secret.iv, "base64"));
    decipher.setAAD(Buffer.from(aad, "utf8"));
    decipher.setAuthTag(Buffer.from(secret.tag, "base64"));
    return Buffer.concat([
      decipher.update(Buffer.from(secret.data, "base64")),
      decipher.final(),
    ]).toString("utf8");
  } catch {
    return null;
  }
}

/* ---------- Helpers ---------- */

function toPublic(row: PublicRow): PublicKey {
  return {
    id: row.id,
    provider: row.provider,
    label: row.label,
    masked: "\u2022".repeat(12) + row.last4,
    enabled: row.enabled,
    createdAt: row.created_at,
    lastResult: row.last_result,
    lastUsedAt: row.last_used_at,
  };
}

function notReady(): { ok: false; code: string; message: string; status: number } {
  return { ok: false, code: "storage_unavailable", message: getStorageState().message, status: 503 };
}

async function fetchRows(provider: ProviderId, ownerId: string): Promise<SecretRow[]> {
  const { data, error } = await db()
    .from(TABLE)
    .select(`${PUBLIC_COLUMNS},secret`)
    .eq("owner_id", ownerId)
    .eq("provider", provider);
  const rows: unknown = data;
  if (error || !Array.isArray(rows) || !rows.every(isSecretRow)) {
    throw new KeyStoreError("The saved keys could not be read.");
  }
  return rows;
}

// Role comes from public.profiles, read here on the server. Never from the caller.
async function isAdminUser(ownerId: string): Promise<boolean> {
  try {
    const { data, error } = await db().from("profiles").select("role").eq("id", ownerId).maybeSingle();
    if (error) {
      return false;
    }
    const row: unknown = data;
    return isRecord(row) && row.role === "admin";
  } catch {
    return false;
  }
}

export async function listKeys(ownerId: string): Promise<PublicKey[]> {
  if (!isOwnerId(ownerId)) {
    throw new KeyStoreError("A signed-in user is required.");
  }
  if (getStorageState().state !== "ready") {
    return [];
  }
  const { data, error } = await db()
    .from(TABLE)
    .select(PUBLIC_COLUMNS)
    .eq("owner_id", ownerId)
    .order("created_at", { ascending: true });
  const rows: unknown = data;
  if (error || !Array.isArray(rows) || !rows.every(isPublicRow)) {
    throw new KeyStoreError("The saved keys could not be read.");
  }
  return rows.map(toPublic);
}

export async function addKey(
  ownerId: string,
  input: { provider: ProviderId; label: string; apiKey: string },
): Promise<KeyResult<PublicKey>> {
  if (!isOwnerId(ownerId)) {
    return notSignedIn();
  }
  if (getStorageState().state !== "ready") {
    return notReady();
  }
  const cryptoKey = getCryptoKey();
  if (!cryptoKey) {
    return notReady();
  }

  // Only this owner's keys are inspected. Another user's keys are never compared.
  const sameProvider = await fetchRows(input.provider, ownerId);

  if (sameProvider.length >= MAX_KEYS_PER_PROVIDER) {
    return {
      ok: false,
      code: "limit_reached",
      status: 409,
      message: `You already have ${MAX_KEYS_PER_PROVIDER} keys for this provider. Remove one first.`,
    };
  }
  if (sameProvider.some((row) => row.label.toLowerCase() === input.label.toLowerCase())) {
    return {
      ok: false,
      code: "duplicate_label",
      status: 409,
      message: "A key with this label already exists for this provider. Choose a different label.",
    };
  }
  for (const existing of sameProvider) {
    if (decrypt(existing.secret, existing.id, cryptoKey) === input.apiKey) {
      return {
        ok: false,
        code: "duplicate_key",
        status: 409,
        message: `This key is already saved as "${existing.label}".`,
      };
    }
  }

  const id = randomUUID();
  const { data, error } = await db()
    .from(TABLE)
    .insert({
      id,
      owner_id: ownerId,
      provider: input.provider,
      label: input.label,
      last4: input.apiKey.slice(-4),
      enabled: true,
      secret: encrypt(input.apiKey, id, cryptoKey),
    })
    .select(PUBLIC_COLUMNS)
    .single();

  if (error) {
    if (error.code === "23505") {
      return {
        ok: false,
        code: "duplicate_label",
        status: 409,
        message: "A key with this label already exists for this provider. Choose a different label.",
      };
    }
    if (error.message.includes("KEY_LIMIT")) {
      return {
        ok: false,
        code: "limit_reached",
        status: 409,
        message: `You already have ${MAX_KEYS_PER_PROVIDER} keys for this provider. Remove one first.`,
      };
    }
    throw new KeyStoreError("The key could not be saved.");
  }

  const row: unknown = data;
  if (!isPublicRow(row)) {
    throw new KeyStoreError("The key was saved but the answer was unexpected.");
  }
  return { ok: true, value: toPublic(row) };
}

export async function setEnabled(
  ownerId: string,
  id: string,
  enabled: boolean,
): Promise<KeyResult<PublicKey>> {
  if (!isOwnerId(ownerId)) {
    return notSignedIn();
  }
  if (getStorageState().state !== "ready") {
    return notReady();
  }
  const { data, error } = await db()
    .from(TABLE)
    .update({ enabled })
    .eq("id", id)
    .eq("owner_id", ownerId)
    .select(PUBLIC_COLUMNS)
    .maybeSingle();
  if (error) {
    throw new KeyStoreError("The key could not be updated.");
  }
  const row: unknown = data;
  if (row === null || row === undefined) {
    return { ok: false, code: "not_found", status: 404, message: "That key no longer exists." };
  }
  if (!isPublicRow(row)) {
    throw new KeyStoreError("The update answer was unexpected.");
  }
  return { ok: true, value: toPublic(row) };
}

export async function removeKey(ownerId: string, id: string): Promise<KeyResult<true>> {
  if (!isOwnerId(ownerId)) {
    return notSignedIn();
  }
  if (getStorageState().state !== "ready") {
    return notReady();
  }
  const { data, error } = await db()
    .from(TABLE)
    .delete()
    .eq("id", id)
    .eq("owner_id", ownerId)
    .select("id");
  if (error) {
    throw new KeyStoreError("The key could not be removed.");
  }
  if (!Array.isArray(data) || data.length === 0) {
    return { ok: false, code: "not_found", status: 404, message: "That key no longer exists." };
  }
  return { ok: true, value: true };
}

/** Records what really happened the last time one of this owner's keys was used. */
export async function recordKeyResult(
  ownerId: string,
  id: string,
  result: Exclude<LastResult, null>,

  }
  try {
    const { error } = await db()
      .from(TABLE)

      .eq("owner_id", ownerId);
    if (error) {

/* ---------- Choosing keys for an AI request ---------- */

function failureRank(row: SecretRow): number {
  if (row.last_result === "rejected") {
    return 2;
  }
  if (
    row.last_result === "rate_limited" &&
    row.last_used_at !== null &&
    Date.now() - Date.parse(row.last_used_at) < RECENT_FAILURE_MS
  ) {
    return 1;
  }
  return 0;
}

function readEnvCandidates(provider: ProviderId): Candidate[] {
  for (const name of ENV_NAMES[provider] ?? []) {
    const value = (process.env[name] ?? "").trim();
    if (value) {
      return [{ id: "", label: `${name} (environment)`, apiKey: value }];
    }
  }
  return [];
}

/**
 * Keys this owner may use for this provider, in the order to try them.
 * - Saved keys of THIS owner only. Enabled ones are used. If all are disabled, nothing is used.
 * - With no saved key for the provider, the environment key is used only when the
 *   server confirms the owner is an admin. Ordinary users never get a shared key.
 */
export async function getCandidates(provider: ProviderId, ownerId: string): Promise<CandidateResult> {
  if (!isOwnerId(ownerId)) {
    return { ok: false, code: "unauthenticated", message: "Please sign in to use AI tools." };
  }

  const providerName = PROVIDERS.find((entry) => entry.id === provider)?.name ?? provider;

  if (getStorageState().state === "ready") {
    let stored: SecretRow[];
    try {
      stored = await fetchRows(provider, ownerId);
    } catch {
      return {
        ok: false,
        code: "key_store_error",
        message:
          "The saved keys could not be read. Check that the api_keys SQL was run and that SUPABASE_SERVICE_ROLE_KEY is correct.",
      };
    }

    if (stored.length > 0) {
      const enabled = stored.filter((row) => row.enabled);
      if (enabled.length === 0) {
        return {
          ok: false,
          code: "no_enabled_key",
          message: `All your saved ${providerName} keys are disabled. Enable one in the API Key Manager.`,
        };
      }

      const cryptoKey = getCryptoKey();
      if (!cryptoKey) {
        return {
          ok: false,
          code: "key_store_error",
          message: "KEY_ENCRYPTION_SECRET is missing on the server.",
        };
      }

      const ordered = [...enabled].sort((a, b) => {
        const rank = failureRank(a) - failureRank(b);
        return rank !== 0 ? rank : Date.parse(a.created_at) - Date.parse(b.created_at);
      });

      const candidates: Candidate[] = [];
      for (const row of ordered) {
        const apiKey = decrypt(row.secret, row.id, cryptoKey);
        if (apiKey) {
          candidates.push({ id: row.id, label: row.label, apiKey });
        }
      }

      if (candidates.length === 0) {
        return {
          ok: false,
          code: "key_store_error",
          message:
            "The saved keys could not be decrypted. KEY_ENCRYPTION_SECRET may have changed. Remove the keys and add them again.",
        };
      }
      return { ok: true, source: "store", candidates };
    }
  }

  if ((ENV_NAMES[provider] ?? []).length > 0 && (await isAdminUser(ownerId))) {
    const fromEnv = readEnvCandidates(provider);
    if (fromEnv.length > 0) {
      return { ok: true, source: "env", candidates: fromEnv };
    }
  }

  return {
    ok: false,
    code: "not_configured",
    message: `No ${providerName} API key is available. Add your own key in the API Key Manager.`,
  };
}

// SERVER ONLY. Picks the CALLING USER's enabled saved keys (or, for the verified admin
// only, the environment key), calls Gemini, and tries at most MAX_KEY_ATTEMPTS keys when
// a key is rejected or rate-limited.
// `userId` must come from the server-verified session, never from the request body.

const MAX_KEY_ATTEMPTS = 3;

const STATUS_BY_CODE: Record<GeminiErrorCode, number> = {
  invalid_key: 401,
  unsupported_model: 400,
  rate_limited: 429,
  provider_unavailable: 503,
  timeout: 504,
  blocked: 422,
  bad_response: 502,
  bad_request: 502,
};

export type GeminiRunInput = {
  userId: string;
  model: string;
  mimeType: string;
  base64: string;
  prompt: string;
  schema?: Record<string, unknown>;
  keywordMax: number;
};

export type GeminiRunResult =
  | { ok: true; raw: unknown }
  | {
      ok: false;
      status: number;
      code: string;
      message: string;
      retryable: boolean;
      retryAfterSeconds: number | null;
    };

function failure(status: number, code: string, message: string, retryable: boolean): GeminiRunResult {
  return { ok: false, status, code, message, retryable, retryAfterSeconds: null };
}

export async function runGeminiWithKeys(input: GeminiRunInput): Promise<GeminiRunResult> {
  // Provider first, then the verified user id (the order used on the multi-provider branch).
  const resolved = await getCandidates("gemini", input.userId);
  if (!resolved.ok) {
    if (resolved.code === "key_store_error") {
      return failure(500, "key_store_error", resolved.message, false);
    }
    // Reported as "not_configured" so the browser stops a whole batch instead of repeating the failure.
    return failure(503, "not_configured", resolved.message, false);
  }

  const attempts = resolved.candidates.slice(0, MAX_KEY_ATTEMPTS);

  for (let index = 0; index < attempts.length; index += 1) {
    const candidate = attempts[index];
    try {
      const raw = await generateRawMetadata({
        apiKey: candidate.apiKey,
        model: input.model,
        mimeType: input.mimeType,
        base64: input.base64,
        prompt: input.prompt,
        keywordMax: input.keywordMax,
        schema: input.schema,
      });
      await recordKeyResult(input.userId, candidate.id, "ok");
      return { ok: true, raw };
    } catch (error) {
      if (error instanceof GeminiError) {
        if (error.code === "invalid_key") {
          await recordKeyResult(input.userId, candidate.id, "rejected");
        } else if (error.code === "rate_limited") {
          await recordKeyResult(input.userId, candidate.id, "rate_limited");
        }

        const keySpecific = error.code === "invalid_key" || error.code === "rate_limited";
        if (keySpecific && index < attempts.length - 1) {
          continue;
        }

        return {
          ok: false,
          status: STATUS_BY_CODE[error.code],
          code: error.code,
          message: error.message,
          retryable: error.retryable,
          retryAfterSeconds: error.retryAfterMs !== null ? Math.ceil(error.retryAfterMs / 1000) : null,
        };
      }

      console.error("[gemini-run] unexpected failure");
      return failure(500, "internal_error", "Something went wrong on the server. Please try again.", true);
    }
  }

  return failure(500, "internal_error", "No key could be used. Please try again.", true);
}

const BASE_URL = "https://generativelanguage.googleapis.com/v1beta/models";
const REQUEST_TIMEOUT_MS = 60_000;
const MAX_RETRIES = 2;
const DEFAULT_BACKOFF_MS = [1500, 4000];
const MAX_INLINE_WAIT_MS = 10_000;
const BLOCKED_FINISH_REASONS = new Set(["SAFETY", "PROHIBITED_CONTENT", "BLOCKLIST", "SPII", "IMAGE_SAFETY"]);

function sleep(ms: number): Promise<void> {
  return new Promise<void>((resolve) => setTimeout(resolve, ms));
}

function isNamed(error: unknown, name: string): boolean {
  return typeof error === "object" && error !== null && (error as { name?: unknown }).name === name;
}

async function readJson(response: Response): Promise<unknown> {
  try {
    return await response.json();
  } catch {
    return null;
  }
}

function readRetryAfterMs(response: Response, detail: unknown): number | null {
  const header = response.headers.get("retry-after");
  if (header) {
    const seconds = Number(header);
    if (Number.isFinite(seconds) && seconds >= 0) {
      return Math.round(seconds * 1000);
    }
  }
  const details = (detail as { error?: { details?: unknown } } | null)?.error?.details;
  if (Array.isArray(details)) {
    for (const entry of details) {
      if (typeof entry === "object" && entry !== null) {
        const value = (entry as { retryDelay?: unknown }).retryDelay;
        if (typeof value === "string") {
          const match = /^(\d+(?:\.\d+)?)s$/.exec(value);
          if (match) {
            return Math.round(Number(match[1]) * 1000);
          }
        }
      }
    }
  }
  return null;
}

function errorFromResponse(
  status: number,
  detail: unknown,
  usedSchema: boolean,
  retryAfterMs: number | null,
): GeminiError {
  const haystack = JSON.stringify(detail ?? {});

  if (status === 400) {
    if (/api key not valid|api_key_invalid/i.test(haystack)) {
      return new GeminiError(
        "invalid_key",
        "Google rejected the Gemini API key. Check that the key is correct and that the Gemini API is enabled for it.",
        false,
      );
    }
    if (usedSchema && /response_?json_?schema|unknown name|schema/i.test(haystack)) {
      throw new SchemaNotAccepted();
    }
    if (/model/i.test(haystack) && /not found|not supported|invalid/i.test(haystack)) {
      return new GeminiError(
        "unsupported_model",
        "Google does not accept this model name for image input. Choose a different model.",
        false,
      );
    }
    return new GeminiError(
      "bad_request",
      "Google rejected the request. Try a different model or a different image.",
      false,
    );
  }

  if (status === 401 || status === 403) {
    return new GeminiError(
      "invalid_key",
      "Google rejected the Gemini API key, or the Gemini API is not enabled for it.",
      false,
    );
  }

  if (status === 404) {
    return new GeminiError(
      "unsupported_model",
      "Google could not find this model. It may have been retired. Choose a different model.",
      false,
    );
  }

  if (status === 429) {
    return new GeminiError(
      "rate_limited",
      "Gemini says the rate limit or quota was reached. Free-tier limits are low. Wait a minute and use Retry failed, or check your quota in Google AI Studio.",
      true,
      retryAfterMs,
    );
  }

  if (status >= 500) {
    return new GeminiError(
      "provider_unavailable",
      "Gemini is temporarily unavailable. Try again in a moment.",
      true,
      retryAfterMs,
    );
  }

  return new GeminiError("bad_request", `Google answered with an unexpected error (${status}).`, false);
}

function extractText(data: unknown): string {
  const root = data as {
    promptFeedback?: { blockReason?: unknown };
    candidates?: Array<{
      finishReason?: string;
      content?: { parts?: Array<{ text?: unknown; thought?: unknown }> };
    }>;
  } | null;

  if (root?.promptFeedback?.blockReason) {
    throw new GeminiError("blocked", "Google's safety filter blocked this image or request.", false);
  }

  const candidate = root?.candidates?.[0];
  if (!candidate) {
    throw new GeminiError("bad_response", "Gemini returned no answer. Try again.", true);
  }

  if (candidate.finishReason && BLOCKED_FINISH_REASONS.has(candidate.finishReason)) {
    throw new GeminiError("blocked", "Google's safety filter blocked this image or request.", false);
  }

  const parts = candidate.content?.parts ?? [];
  const text = parts
    .filter((part) => typeof part.text === "string" && part.thought !== true)
    .map((part) => part.text as string)
    .join("")
    .trim();

  if (!text) {
    const reason =
      candidate.finishReason === "MAX_TOKENS"
        ? "Gemini's answer was cut off. Try again."
        : "Gemini returned an empty answer. Try again.";
    throw new GeminiError("bad_response", reason, true);
  }

  return text;
}

function parseJsonText(text: string): unknown {
  const cleaned = text
    .trim()
    .replace(/^```(?:json)?\s*/i, "")
    .replace(/\s*```$/, "");
  try {
    return JSON.parse(cleaned);
  } catch {
    throw new GeminiError("bad_response", "Gemini did not return valid JSON. Try again.", true);
  }
}

async function requestText(args: GeminiRequest, useSchema: boolean): Promise<string> {
  const url = `${BASE_URL}/${encodeURIComponent(args.model)}:generateContent`;

  const generationConfig: Record<string, unknown> = {
  responseMimeType: "application/json",
  maxOutputTokens: 8192,
  temperature: 0.2,
};
  if (useSchema) {
    generationConfig.responseJsonSchema = args.schema ?? buildResponseSchema(args.keywordMax);
  }

  const body = {
    contents: [
      {
        role: "user",
        parts: [
          { inline_data: { mime_type: args.mimeType, data: args.base64 } },
          { text: args.prompt },
        ],
      },
    ],
    generationConfig,
  };

  let response: Response;
  try {
    response = await fetch(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-goog-api-key": args.apiKey,
      },
      body: JSON.stringify(body),
      signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS),
      cache: "no-store",
    });
  } catch (error) {
    if (isNamed(error, "TimeoutError") || isNamed(error, "AbortError")) {
      throw new GeminiError("timeout", "Gemini took too long to answer. Try again.", true);
    }
    throw new GeminiError(
      "provider_unavailable",
      "Could not reach Google's servers. Check your internet connection and try again.",
      true,
    );
  }

  if (!response.ok) {
    const detail = await readJson(response);
    // Log only the status code. Never log the key, the image or Google's full message.
    console.error(`[gemini] request failed with status ${response.status}`);
    throw errorFromResponse(response.status, detail, useSchema, readRetryAfterMs(response, detail));
  }

  return extractText(await readJson(response));
}

/**
 * Calls Gemini and returns the parsed JSON.
 * Bounded: at most one schema fallback and at most MAX_RETRIES retries.
 */
export async function generateRawMetadata(args: GeminiRequest): Promise<unknown> {
  let useSchema = true;
  let attempt = 0;

  for (;;) {
    try {
      return parseJsonText(await requestText(args, useSchema));
    } catch (error) {
      if (error instanceof SchemaNotAccepted && useSchema) {
        useSchema = false;
        continue;
      }

      if (error instanceof GeminiError && error.retryable && attempt < MAX_RETRIES) {
        const wait = error.retryAfterMs ?? DEFAULT_BACKOFF_MS[attempt] ?? 4000;
        if (wait > MAX_INLINE_WAIT_MS) {
          throw error;
        }
        await sleep(wait);
        attempt += 1;
        continue;
      }

      if (error instanceof SchemaNotAccepted) {
        throw new GeminiError("bad_request", "Google rejected the request. Try a different model.", false);
      }

      throw error;
    }
  }
}

export type StrictOutcome =
  | { ok: true; data: StrictData; warnings: string[] }
  | {
      ok: false;
      status: number;
      code: string;
      message: string;
      retryable: boolean;
      retryAfterSeconds: number | null;

export async function generateStrictMetadata(input: {
  /** Server-verified user ID from the session. Never from the browser. */
  userId: string;
  model: string;
  mimeType: string;
  base64: string;
  count: number;
}): Promise<StrictOutcome> {
  let problems: string[] = [];

  for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt += 1) {
    const run = await runGeminiWithKeys({
      userId: input.userId,
      model: input.model,
      mimeType: input.mimeType,
      base64: input.base64,
      prompt: buildStrictPrompt(input.count, problems),
      schema: buildStrictSchema(input.count),
      keywordMax: input.count,
    });

    if (!run.ok) {
      return {
        ok: false,
        status: run.status,
        code: run.code,
        message: run.message,
        retryable: run.retryable,
        retryAfterSeconds: run.retryAfterSeconds,
      };
    }

    const checked = validateStrict(run.raw, input.count);
    if (checked.ok) {
      const warnings =
        attempt > 1 ? [...checked.warnings, `Needed ${attempt} attempts to meet the rules.`] : checked.warnings;
      return { ok: true, data: checked.data, warnings };
    }
    problems = checked.problems;
  }

  return {
    ok: false,
    status: 502,
    code: "validation_failed",
    message: `The model could not meet the metadata rules after ${MAX_ATTEMPTS} attempts: ${problems.join(" ")}`,
    retryable: true,
    retryAfterSeconds: null,
  };
}

function toEntry(value: unknown): HistoryEntry | null {
  if (
    !isRecord(value) ||
    typeof value.id !== "string" ||
    (value.kind !== "metadata" && value.kind !== "prompt") ||
    typeof value.file_name !== "string" ||
    typeof value.provider !== "string" ||

    !isRecord(value.result) ||
    typeof value.created_at !== "string"
  ) {
    return null;
  }
  return {
    id: value.id,
    kind: value.kind,
    fileName: value.file_name,
    provider: value.provider,
    model: value.model,
    platform: value.platform,
    result: value.result,
    createdAt: value.created_at,

export async function listHistory(
  userId: string,
  options: { kind: HistoryKind | null; page: number },
): Promise<{ entries: HistoryEntry[]; total: number }> {
  const from = (options.page - 1) * HISTORY_PAGE_SIZE;
  const to = from + HISTORY_PAGE_SIZE - 1;

  try {
    const supabase = await createClient();
    let query = supabase
      .from("history")
      .select(COLUMNS, { count: "exact" })
      .eq("user_id", userId)
      .order("created_at", { ascending: false })
      .range(from, to);
    if (options.kind) {
      query = query.eq("kind", options.kind);
    }

    const { data, error, count } = await query;
    const rows: unknown = data;
    if (error || !Array.isArray(rows)) {
      throw new HistoryError("History could not be read.");
    }

    const entries: HistoryEntry[] = [];
    for (const row of rows) {
      const entry = toEntry(row);
      if (entry) {
        entries.push(entry);
      }
    }
    return { entries, total: count ?? entries.length };
  } catch (error) {
    if (error instanceof HistoryError) {
      throw error;
    }
    throw new HistoryError("History could not be read.");
  }
}

export async function deleteHistory(userId: string, id: string): Promise<boolean> {
  try {
    const supabase = await createClient();
    const { data, error } = await supabase
      .from("history")
      .delete()
      .eq("id", id)
      .eq("user_id", userId)
      .select("id");
    if (error) {
      throw new HistoryError("The entry could not be deleted.");
    }
    const rows: unknown = data;
    return Array.isArray(rows) && rows.length > 0;
  } catch (error) {
    if (error instanceof HistoryError) {
      throw error;
    }
    throw new HistoryError("The entry could not be deleted.");
  }
}

// Strict metadata rules: title <= 70 chars, one-sentence description <= 180 chars,
// exactly `count` unique keywords (count comes from the UI and platform rules).

export const STRICT_LIMITS = { titleMax: 70, descriptionMax: 180 } as const;

export type StrictData = { title: string; description: string; keywords: string[] };

export type StrictValidation =
  | { ok: true; data: StrictData; warnings: string[] }
  | { ok: false; problems: string[] };

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

export function cleanKeyword(raw: string): string {
  return raw
    .replace(/^\s*(?:[-*\u2022]|\d+[.)])\s*/, "")
    .replace(/^["'`]+|["'`]+$/g, "")
    .replace(/[.;:]+$/, "")
    .replace(/\s+/g, " ")
    .trim();
}

export function normalizeKeywords(value: unknown): string[] {
  const parts: string[] = Array.isArray(value)
    ? value.filter((item): item is string => typeof item === "string")
    : typeof value === "string"
      ? value.split(",")
      : [];

  const seen = new Set<string>();
  const result: string[] = [];
  for (const part of parts) {
    const keyword = cleanKeyword(part);
    if (keyword.length === 0 || keyword.length > 60) {
      continue;
    }
    const key = keyword.toLowerCase();
    if (seen.has(key)) {
      continue;
    }
    seen.add(key);
    result.push(keyword);
  }
  return result;
}

function cleanText(value: unknown, label: string): string {
  if (typeof value !== "string") {
    return "";
  }
  const withoutLabel = value.replace(new RegExp(`^\\s*${label}\\s*:\\s*`, "i"), "");
  return withoutLabel.replace(/^["'`]+|["'`]+$/g, "").replace(/\s+/g, " ").trim();
}

function sentenceCount(text: string): number {
  return text.split(/[.!?]+\s+/).filter((part) => part.trim().length > 0).length;
}

export function validateStrict(raw: unknown, count: number): StrictValidation {
  if (!isRecord(raw)) {
    return { ok: false, problems: ["The response was not a JSON object."] };
  }

  const problems: string[] = [];
  const warnings: string[] = [];

  const title = cleanText(raw.title, "title");
  if (title.length === 0) {
    problems.push("The title is missing.");
  } else if (title.length > STRICT_LIMITS.titleMax) {
    problems.push(`The title is ${title.length} characters; the maximum is ${STRICT_LIMITS.titleMax}.`);
  }

  const description = cleanText(raw.description, "description");
  if (description.length === 0) {
    problems.push("The description is missing.");
  } else {
    if (description.length > STRICT_LIMITS.descriptionMax) {
      problems.push(
        `The description is ${description.length} characters; the maximum is ${STRICT_LIMITS.descriptionMax}.`,
      );
    }
    if (sentenceCount(description) !== 1) {
      problems.push("The description must be exactly one sentence.");
    }
    if (title.length > 0 && description.toLowerCase() === title.toLowerCase()) {
      problems.push("The description repeats the title.");
    }
  }

  let keywords = normalizeKeywords(raw.keywords);
  if (keywords.length < count) {
    problems.push(`Only ${keywords.length} unique keywords were returned; exactly ${count} are required.`);
  } else if (keywords.length > count) {
    keywords = keywords.slice(0, count);
    warnings.push(`Extra keywords were removed so exactly ${count} remain.`);
  }

  if (problems.length > 0) {
    return { ok: false, problems };
  }
  return { ok: true, data: { title, description, keywords }, warnings };
}

export function buildStrictSchema(count: number): Record<string, unknown> {
  return {
    type: "object",
    properties: {
      title: { type: "string", maxLength: STRICT_LIMITS.titleMax },
      description: { type: "string", maxLength: STRICT_LIMITS.descriptionMax },
      keywords: { type: "array", minItems: count, maxItems: count, items: { type: "string" } },
    },
    required: ["title", "description", "keywords"],
  };
}

export function buildStrictPrompt(count: number, problems: string[] = []): string {
  const lines = [
    "You write stock-image metadata for the single image provided.",
    "",
    "Rules:",
    `- title: one accurate, natural, professional English title, at most ${STRICT_LIMITS.titleMax} characters including spaces. No quotation marks, no label, no promotional language.`,
    `- description: exactly ONE complete sentence, at most ${STRICT_LIMITS.descriptionMax} characters including spaces, describing the visible subject, composition, background, colors and style. Do not repeat the title.`,
    `- keywords: an array of EXACTLY ${count} unique English keywords, most important subject and concepts first.`,
    "- Describe only what is clearly visible. Never invent objects, brands, people or places.",
    "- Keywords may cover objects, colors, composition, style, background, mood and use cases, and meaningful phrases such as copy space or flat lay are allowed.",
    "- No duplicates, no trivial variations of the same word, no unrelated filler.",
    "",
    "Return only a JSON object with the fields title, description and keywords.",
  ];
  if (problems.length > 0) {
    lines.push(
      "",
      "Your previous answer was rejected for these reasons:",
      ...problems.map((problem) => `- ${problem}`),
      `Fix every problem. The keywords array must contain exactly ${count} unique, relevant entries.`,
    );
  }
  return lines.join("\n");
}

export function buildPrompt(profile: PlatformProfile, keywordCount: number): string {
  const lines = [

function cleanText(value: unknown): string {
  return typeof value === "string" ? value.replace(/\s+/g, " ").trim() : "";
}

function shortenToLimit(text: string, max: number): string {
  if (text.length <= max) {
    return text;
  }
  const cut = text.slice(0, max);
  const lastSpace = cut.lastIndexOf(" ");
  const base = lastSpace >= Math.floor(max * 0.6) ? cut.slice(0, lastSpace) : cut;
  return base.replace(/[\s,;:-]+$/, "");
}

function readKeywords(value: unknown): string[] {
  let source: unknown[] = [];
  if (Array.isArray(value)) {
    source = value;
  } else if (typeof value === "string") {
    source = value.split(",");
  }

  const seen = new Set<string>();
  const result: string[] = [];
  for (const item of source) {
    const keyword = cleanText(item).replace(/^["'\u201C\u201D]+|["'\u201C\u201D]+$/g, "");
    if (!keyword || keyword.length > MAX_KEYWORD_LENGTH) {
      continue;
    }
    const lower = keyword.toLowerCase();
    if (seen.has(lower)) {
      continue;
    }
    seen.add(lower);
    result.push(keyword);
  }
  return result;
}

/** Checks the model's JSON, cleans it up and applies the platform profile. */
export function normalizeMetadata(
  raw: unknown,
  profile: PlatformProfile,
  keywordCount: number,
): NormalizeResult {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    return { ok: false, message: "The model did not return a JSON object." };
  }

  const record = raw as Record<string, unknown>;
  const warnings: string[] = [];

  let title = cleanText(record.title);
  let description = cleanText(record.description);
  let keywords = readKeywords(record.keywords);

  if (!title) {
    return { ok: false, message: "The model response had no title." };
  }
  if (!description) {
    return { ok: false, message: "The model response had no description." };
  }
  if (keywords.length === 0) {
    return { ok: false, message: "The model response had no usable keywords." };
  }

  if (title.length > profile.titleMaxChars) {
    title = shortenToLimit(title, profile.titleMaxChars);
    warnings.push(`The title was shortened to fit ${profile.titleMaxChars} characters. Please check it still reads well.`);
  }
  if (description.length > profile.descriptionMaxChars) {
    description = shortenToLimit(description, profile.descriptionMaxChars);
    warnings.push(`The description was shortened to fit ${profile.descriptionMaxChars} characters. Please check it still reads well.`);
  }
  if (keywords.length > keywordCount) {
    keywords = keywords.slice(0, keywordCount);
    warnings.push(`Extra keywords were removed to match the requested ${keywordCount}.`);
  }
  if (profile.keywordsMin !== null && keywords.length < profile.keywordsMin) {
    warnings.push(
      `Only ${keywords.length} keywords were returned. ${profile.name} is reported to want at least ${profile.keywordsMin}. Add a few accurate ones by hand if needed.`,
    );
  }

  return { ok: true, data: { title, description, keywords }, warnings };
}

export type PromptNormalizeResult =
  | { ok: true; prompt: string; warnings: string[] }
  | { ok: false; message: string };

export function normalizePrompt(raw: unknown, detail: PromptDetailOption): PromptNormalizeResult {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    return { ok: false, message: "The model did not return a JSON object." };
  }

  const value = (raw as Record<string, unknown>).prompt;
  let prompt = typeof value === "string" ? value.replace(/\s+/g, " ").trim() : "";
  if (prompt.length < 20) {
    return { ok: false, message: "The model response had no usable prompt." };
  }

  const warnings: string[] = [];
  if (prompt.length > detail.maxChars) {
    const cut = prompt.slice(0, detail.maxChars);
    const lastStop = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf(", "), cut.lastIndexOf(" "));
    prompt = (lastStop > detail.maxChars * 0.6 ? cut.slice(0, lastStop) : cut).replace(/[\s,;:-]+$/, "");
    warnings.push(`The prompt was shortened to fit ${detail.maxChars} characters. Please check it still reads well.`);
  }

  return { ok: true, prompt, warnings };
}

export async function POST(request: Request): Promise<Response> {
  const requestId = randomUUID();

  if (!isSameOrigin(request)) {
    return jsonError(403, "forbidden", "This request was not allowed.", false, requestId);
  }

  const client = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "local";
  const limit = checkRateLimit(`generate:${client}`, RATE_LIMIT_PER_MINUTE, 60_000);
  if (!limit.allowed) {

  } catch {
    return jsonError(400, "invalid_request", "The upload could not be read.", false, requestId);

  const platform = formField(form, "platform");
  const requested = Number(formField(form, "keywordCount"));

  }

    return jsonError(400, "invalid_request", "Unknown platform.", false, requestId);
  }
  if (!Number.isInteger(requested) || requested < 1 || requested > 50) {
    return jsonError(400, "invalid_request", "The keyword count must be a whole number from 1 to 50.", false, requestId);
  }

  if (!upload.ok) {
    return upload.response;
  }

    return jsonError(
      generated.status,
      generated.code,
      generated.message,
      generated.retryable,

        ? { "Retry-After": String(generated.retryAfterSeconds) }
        : undefined,
    );

  const warnings = generated.warnings;

    kind: "metadata",
    fileName: cleanFileName(formField(form, "fileName")),
    provider: "gemini",
    model,
    platform: profile.id,
    result: { title, description, keywords, warnings },
  });

    {
      success: true,
      data: {
        title,
        description,
        keywords: keywords.join(", "), // CHECK: use `keywords` (array) if the UI expects an array
        warnings,
      },

      provider: "gemini",
      model,
      platform: profile.id,
